server {
    listen       8089;
    server_name  localhost;

    location / {
        root   C:\部署\my-web;
        index  index.html index.htm;
        try_files $uri $uri/ /index.html;

        # 只拦截POST/OPTIONS等接口请求，GET首页不做来源校验
        if ($request_method !~ ^GET$ ) {
            set $allow_source 0;
            # 匹配Origin
            if ($http_origin ~* "^http://1\.94\.18\.1:8089$") {
                set $allow_source 1;
            }
            # 匹配Referer
            if ($http_referer ~* "^http://1\.94\.18\.1:8089") {
                set $allow_source 1;
            }
            # 非白名单接口拦截403
            if ($allow_source = 0) {
                return 403;
            }
        }

        # 跨域响应头
        add_header Access-Control-Allow-Origin "http://1.94.18.1:8089";
        add_header Access-Control-Allow-Methods GET,POST,OPTIONS,PUT,DELETE;
        add_header Access-Control-Allow-Headers *;
    }

    # OPTIONS预检放行
    if ($request_method = 'OPTIONS') {
        return 204;
    }

    error_page   500 502 503 504  /50x.html;
    location = /50x.html {
        root   html;
    }
}
